#
# ГЛАВНОЕ ПРАВИЛО, КОТОРОЕ ЧАЩЕ ВСЕГО НАРУШАЮТ:
#
#   robots.txt управляет ОБХОДОМ, а не индексацией.
#   Disallow не убирает страницу из выдачи — он запрещает её скачивать.
#   Если на закрытую страницу есть внешние ссылки, она всё равно попадёт
#   в индекс, только без заголовка и описания: «Описание недоступно
#   из-за ограничений в robots.txt».
#
#   Чтобы страница НЕ индексировалась, нужен <meta name="robots" content="noindex">
#   в её <head> — и страница при этом должна быть ОТКРЫТА для обхода,
#   иначе робот не сможет прочитать этот мета-тег.
#
#   Именно поэтому /spasibo/ ниже НЕ закрыт: у неё стоит noindex в HTML.
#   Закроете её здесь — она попадёт в выдачу вместо посадочных.
#
# ------------------------------------------------------------------------------
# ЧЕГО ЗДЕСЬ НАМЕРЕННО НЕТ:
#
#   Host:        директива отменена Яндексом в 2018 году и игнорируется.
#                Главное зеркало задаётся 301-редиректом и canonical.
#
#   Crawl-delay: Яндекс перестал её учитывать в 2018-м, скорость обхода
#                настраивается в Яндекс.Вебмастере. Google её не понимал никогда.
#
# ------------------------------------------------------------------------------
# ЗАГЛУШКИ ПЕРЕД ПУБЛИКАЦИЕЙ:
#   1. Заменить https://example.ru на реальный адрес с https
#   2. Проверить в Яндекс.Вебмастере: Инструменты → Анализ robots.txt
#   3. Проверить в Google Search Console: Настройки → robots.txt
# ==============================================================================


User-agent: *

# Разрешаем весь сайт
Allow: /

# Служебные разделы. ЗАГЛУШКА: оставьте только те, что реально существуют
# на вашем хостинге — лишние строки ничего не ломают, но засоряют файл.
Disallow: /admin/
Disallow: /cgi-bin/
Disallow: /tmp/
Disallow: /*.log$

# Результаты внутреннего поиска и сортировки — если появятся.
# Такие страницы плодят дубли и тратят краулинговый бюджет.
Disallow: /search
Disallow: /*?s=
Disallow: /*?sort=

# Обработчик форм, если перейдёте на свой скрипт вместо Formspree
Disallow: /send.php

# ВНИМАНИЕ: /spasibo/ здесь НЕ закрыта — см. пояснение вверху файла.
# У неё стоит <meta name="robots" content="noindex"> в HTML, и робот
# должен иметь возможность его прочитать.

# Ресурсы, нужные для корректного рендеринга страницы роботом.
# Если закрыть CSS/JS/шрифты, Google увидит сайт «сломанным» и может
# занизить оценку мобильной версии.
Allow: /assets/
Allow: /fonts/
Allow: /img/
Allow: /*.css$
Allow: /*.js$
Allow: /*.woff2$


# ------------------------------------------------------------------------------
# Clean-param — только для Яндекса.
# Склеивает адреса с рекламными метками в один канонический URL,
# чтобы /remont-holodilnikov/surgut/?utm_source=yandex не считался
# отдельной страницей. Для Google ту же задачу решает <link rel="canonical">,
# который уже стоит на всех страницах.
#
# Синтаксис: Clean-param: параметр1&параметр2 [путь]
# ------------------------------------------------------------------------------
User-agent: Yandex
Allow: /
Clean-param: utm_source&utm_medium&utm_campaign&utm_term&utm_content
Clean-param: yclid&gclid&fbclid&ysclid&_openstat&from&roistat
